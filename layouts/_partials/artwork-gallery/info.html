{{- /* Everything the templates need to know about one artwork page. Call with partialCached "artwork-gallery/info.html" $page $page.RelPermalink */ -}} {{- $cfg := partialCached "artwork-gallery/config.html" . -}} {{- $artist := site.Params.artist | default site.Title -}} {{- $title := .Title -}} {{- $status := .Params.status | default "" | lower -}} {{- $displayTitle := $title | default $cfg.untitled -}} {{- $alt := .Params.alt | default $title | default (printf "%s by %s" $cfg.artform $artist) -}} {{- /* Untitled pieces and repeated titles get their gallery number so every page has a unique browser and search title. */ -}} {{- $counts := partialCached "artwork-gallery/title-counts.html" . -}} {{- $repeated := or (not $title) (gt (index $counts (lower $title)) 1) -}} {{- $number := string .Weight -}} {{- if not .Weight -}}{{- $number = .Date.Format "2006-01-02" -}}{{- end -}} {{- $seoTitle := cond $repeated (printf "%s — Work %s" $displayTitle $number) $title -}} {{- $seoName := cond $repeated (printf "%s, gallery work %s" $displayTitle $number) $title -}} {{- $description := printf "%s by %s." $seoName $artist -}} {{- with $status -}}{{- $description = printf "%s %s." $description . -}}{{- end -}} {{- /* The artwork image: front matter `image`, else artwork.*, else the first image in the bundle that is not the sharing card. */ -}} {{- $src := false -}} {{- with .Params.image -}}{{- $src = $.Resources.Get . -}}{{- end -}} {{- if not $src -}}{{- $src = .Resources.GetMatch "artwork.*" -}}{{- end -}} {{- if not $src -}}{{- with partial "artwork-gallery/images.html" . -}}{{- $src = index . 0 -}}{{- end -}}{{- end -}} {{- if not $src -}} {{- errorf "artwork-gallery: %s has no artwork image in its folder" .File.Path -}} {{- end -}} {{- $thumbSpec := printf "fill %dx%d Center" $cfg.thumbSize $cfg.thumbSize -}} {{- $largeSpec := printf "fit %dx%d" $cfg.largeSize $cfg.largeSize -}} {{- $thumb := $src.Filter images.AutoOrient (images.Process (printf "%s q%d jpg" $thumbSpec $cfg.thumbQuality)) -}} {{- $thumbWebp := $src.Filter images.AutoOrient (images.Process (printf "%s q%d webp" $thumbSpec $cfg.webpThumbQuality)) -}} {{- $large := $src.Filter images.AutoOrient (images.Process (printf "%s q%d jpg" $largeSpec $cfg.largeQuality)) -}} {{- $largeWebp := $src.Filter images.AutoOrient (images.Process (printf "%s q%d webp" $largeSpec $cfg.webpLargeQuality)) -}} {{- /* Other photos in the folder are further views of the piece, in the folder's display order, each with a caption. */ -}} {{- $views := slice -}} {{- range partial "artwork-gallery/images.html" . -}} {{- if ne .Name $src.Name -}} {{- $caption := partial "artwork-gallery/caption.html" . -}} {{- $views = $views | append (dict "caption" $caption "alt" (.Params.alt | default (cond (ne $caption "") (printf "%s — %s" $displayTitle $caption) $alt)) "thumb" (.Filter images.AutoOrient (images.Process (printf "%s q%d jpg" $thumbSpec $cfg.thumbQuality))) "thumbWebp" (.Filter images.AutoOrient (images.Process (printf "%s q%d webp" $thumbSpec $cfg.webpThumbQuality))) "large" (.Filter images.AutoOrient (images.Process (printf "%s q%d jpg" $largeSpec $cfg.largeQuality))) "largeWebp" (.Filter images.AutoOrient (images.Process (printf "%s q%d webp" $largeSpec $cfg.webpLargeQuality))) ) -}} {{- end -}} {{- end -}} {{- /* Sharing card: social.* when the folder has one, otherwise the artwork uncropped on a 1200 x 630 field of its own dominant colour. */ -}} {{- $social := .Resources.GetMatch "social.*" -}} {{- if not $social -}} {{- $fit := $src.Filter images.AutoOrient (images.Process "fit 1120x560 q85 jpg") -}} {{- $colour := index $src.Colors 0 | default "#ffffff" -}} {{- $left := div (sub 1200 $fit.Width) 2 -}} {{- $top := div (sub 630 $fit.Height) 2 -}} {{- $pad := images.Padding $top (sub (sub 1200 $fit.Width) $left) (sub (sub 630 $fit.Height) $top) $left $colour -}} {{- $social = $fit.Filter $pad -}} {{- end -}} {{- return dict "page" . "title" $title "displayTitle" $displayTitle "status" $status "year" (.Params.year | default "") "alt" $alt "seoTitle" $seoTitle "documentTitle" (printf "%s | %s" $seoTitle $artist) "description" $description "thumb" $thumb "thumbWebp" $thumbWebp "large" $large "source" $src "largeWebp" $largeWebp "social" $social "views" $views -}}
