{{- /* The artwork pages directly inside one collection, in gallery order (weight, then date, then title). Call with partialCached "artwork-gallery/artworks.html" $collection $collection.RelPermalink */ -}} {{- $pages := slice -}} {{- range where .RegularPages.ByWeight "Type" "artworks" -}} {{- if eq (partialCached "artwork-gallery/kind.html" . .RelPermalink) "artwork" -}} {{- $pages = $pages | append . -}} {{- end -}} {{- end -}} {{- $seen := dict -}} {{- range $p := $pages -}} {{- with $p.Weight -}} {{- $key := string . -}} {{- with index $seen $key -}} {{- errorf "artwork-gallery: %s and %s both use weight %s" . $p.File.Path $key -}} {{- end -}} {{- $seen = merge $seen (dict $key $p.File.Path) -}} {{- end -}} {{- end -}} {{- return $pages -}}
