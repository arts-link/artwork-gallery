{{- /* The public pieces shown in one collection, in order. Call with partialCached "artwork-gallery/artworks.html" $collection $collection.RelPermalink - the gallery section: every piece, by weight, then date, then title - a term of a collection taxonomy: its pieces by `_weight` (e.g. sculpture_weight), then weight; pieces without one go last - a collection taxonomy page: none; it shows its terms as cards */ -}} {{- $candidates := slice -}} {{- if eq .Kind "term" -}} {{- $keyed := slice -}} {{- with index site.Taxonomies .Data.Plural -}} {{- with index . $.Data.Term -}} {{- range . -}} {{- $series := cond (eq .Weight 0) 999999 .Weight -}} {{- $own := cond (eq .Page.Weight 0) 999999 .Page.Weight -}} {{- $keyed = $keyed | append (dict "page" .Page "key" (add (mul $series 1000000) $own)) -}} {{- end -}} {{- end -}} {{- end -}} {{- range sort $keyed "key" -}}{{- $candidates = $candidates | append .page -}}{{- end -}} {{- else if .IsSection -}} {{- $candidates = .RegularPages.ByWeight -}} {{- end -}} {{- $pages := slice -}} {{- range where $candidates "Type" "artworks" -}} {{- if not .Params.private -}}{{- $pages = $pages | append . -}}{{- end -}} {{- end -}} {{- /* Two pieces with the same weight still build; the order between them falls back to date and title, and the build log says so. */ -}} {{- if .IsSection -}} {{- $seen := dict -}} {{- range $p := $pages -}} {{- with $p.Weight -}} {{- $key := string . -}} {{- with index $seen $key -}} {{- warnf "artwork-gallery: %s and %s both use weight %s" . $p.File.Path $key -}} {{- end -}} {{- $seen = merge $seen (dict $key $p.File.Path) -}} {{- end -}} {{- end -}} {{- end -}} {{- return $pages -}}
