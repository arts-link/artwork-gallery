{{- /* Split the gallery into content-addressed JSON batches so later artwork data is only downloaded as a visitor scrolls. Returns the records for the first (server-rendered) batch and the URLs of the rest. */ -}} {{- /* Call with partialCached "artwork-gallery/batches.html" $collection $collection.RelPermalink */ -}} {{- $cfg := partialCached "artwork-gallery/config.html" . -}} {{- $size := int $cfg.batchSize -}} {{- $records := slice -}} {{- $dir := printf "assets/data/%s" (strings.Trim .RelPermalink "/") -}} {{- range $i, $p := partialCached "artwork-gallery/artworks.html" . .RelPermalink -}} {{- $records = $records | append (partial "artwork-gallery/record.html" (dict "page" $p "index" $i)) -}} {{- end -}} {{- $urls := slice -}} {{- $n := 0 -}} {{- if gt (len $records) $size -}} {{- range $start := seq $size $size (sub (len $records) 1) -}} {{- $n = add $n 1 -}} {{- $json := $records | after $start | first $size | jsonify -}} {{- $batch := resources.FromString (printf "%s/batch-%d.json" $dir $n) (printf "%s\n" $json) | fingerprint "sha256" -}} {{- $urls = $urls | append $batch.RelPermalink -}} {{- end -}} {{- end -}} {{- return dict "first" (first $size $records) "urls" $urls "count" (len $records) -}}
