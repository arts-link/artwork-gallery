{{- /* The resized versions of one image for a : AVIF and WebP srcsets at the requested widths (never wider than the source), plus a JPEG for old browsers, structured data and sharing. With "max", no size is wider than that. With "native" true (piece images), the source's own width (or "max", if smaller) is added as the largest size when it is at least 10% wider than the largest listed width it reaches, so no piece stops short of its full resolution. Sizes are measured after EXIF orientation, so a photo stored sideways is never upscaled. Call with partialCached "artwork-gallery/derivatives.html" (dict "image" "widths" "square" "native" "max" "fallback" ) Quality and encoder settings come from the site's [imaging.avif], [imaging.webp] and [imaging.jpeg] configuration. */ -}} {{- $img := .image -}} {{- $width := $img.Width -}} {{- $height := $img.Height -}} {{- with $img.Meta -}} {{- if ge (int .Orientation) 5 -}}{{- $width = $img.Height -}}{{- $height = $img.Width -}}{{- end -}} {{- end -}} {{- $limit := $width -}} {{- if .square -}}{{- $limit = math.Min $width $height -}}{{- end -}} {{- $cap := $limit -}} {{- with .max -}}{{- $cap = math.Min $limit . -}}{{- end -}} {{- $widths := slice -}} {{- range .widths -}}{{- if le . $cap -}}{{- $widths = $widths | append . -}}{{- end -}}{{- end -}} {{- if not $widths -}}{{- $widths = slice (int $cap) -}}{{- end -}} {{- if and .native (not .square) -}} {{- $largest := index $widths (sub (len $widths) 1) -}} {{- if ge (int $cap) (mul $largest 1.1) -}}{{- $widths = $widths | append (int $cap) -}}{{- end -}} {{- end -}} {{- $spec := "resize %dx %s" -}} {{- if .square -}}{{- $spec = "fill %dx%[1]d Center %s" -}}{{- end -}} {{- $sets := dict -}} {{- range $format := slice "avif" "webp" -}} {{- $candidates := slice -}} {{- range $widths -}} {{- $r := $img.Filter images.AutoOrient (images.Process (printf $spec . $format)) -}} {{- $candidates = $candidates | append (printf "%s %dw" $r.RelPermalink $r.Width) -}} {{- end -}} {{- $sets = merge $sets (dict $format (delimit $candidates ", ")) -}} {{- end -}} {{- $fallback := math.Min (.fallback | default 1200) $limit | int -}} {{- $jpg := $img.Filter images.AutoOrient (images.Process (printf $spec $fallback "jpg")) -}} {{- return dict "avif" $sets.avif "webp" $sets.webp "jpg" $jpg "width" $jpg.Width "height" $jpg.Height -}}
