{{- /* The small, browser-safe description of one gallery tile. The first batch is rendered as HTML; later batches are the same records as JSON. Context: dict "page" "index" "collection" */ -}} {{- $cfg := partialCached "artwork-gallery/config.html" . -}} {{- $info := partialCached "artwork-gallery/info.html" .page .page.RelPermalink -}} {{- $record := dict "i" .index "slug" .page.File.ContentBaseName "href" .page.RelPermalink "image" $info.largeWebp.RelPermalink "title" $info.title "meta" $info.status "alt" $info.alt "documentTitle" $info.documentTitle "thumbWebp" $info.thumbWebp.RelPermalink "thumb" $info.thumb.RelPermalink -}} {{- with $info.views -}} {{- $views := slice (dict "src" $info.largeWebp.RelPermalink "thumb" $info.thumbWebp.RelPermalink "caption" "" "alt" $info.alt) -}} {{- range . -}} {{- $views = $views | append (dict "src" .largeWebp.RelPermalink "thumb" .thumbWebp.RelPermalink "caption" .caption "alt" .alt) -}} {{- end -}} {{- $record = merge $record (dict "views" $views) -}} {{- end -}} {{- /* A justified grid keeps each image's shape instead of the square crop. */ -}} {{- if eq (.collection.Params.grid | default $cfg.grid) "justified" -}} {{- $spec := printf "fit %dx%d" (div (mul $cfg.thumbSize 3) 2) (div (mul $cfg.thumbSize 3) 2) -}} {{- $jpg := $info.source.Filter images.AutoOrient (images.Process (printf "%s q%d jpg" $spec $cfg.thumbQuality)) -}} {{- $webp := $info.source.Filter images.AutoOrient (images.Process (printf "%s q%d webp" $spec $cfg.webpThumbQuality)) -}} {{- $record = merge $record (dict "thumb" $jpg.RelPermalink "thumbWebp" $webp.RelPermalink "w" $jpg.Width "h" $jpg.Height) -}} {{- end -}} {{- return $record -}}
